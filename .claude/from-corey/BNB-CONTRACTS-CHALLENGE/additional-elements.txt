solidity/**
 * @notice Get current price in BNB per token (scaled by 1e18)
 * @return Current price with 18 decimals
 */
function getCurrentPrice() external view returns (uint256) {
    uint256 currentBnb = bnbReserves + VIRTUAL_BNB_RESERVES;
    uint256 currentTokens = balanceOf(address(this)) + VIRTUAL_TOKEN_RESERVES;
    
    // Price = BNB / Token, scaled to 18 decimals
    return (currentBnb * 1e18) / currentTokens;
}

6. Mathematical Edge Cases and Precision
6.1 Handling Division Precision
solidity// ISSUE: Integer division truncates
// SOLUTION: Use high-precision intermediate calculations

function getAmountOfTokens(uint256 bnbAmount) internal view returns (uint256) {
    uint256 currentBnb = bnbReserves + VIRTUAL_BNB_RESERVES;
    uint256 currentTokens = balanceOf(address(this)) + VIRTUAL_TOKEN_RESERVES;
    
    // Calculate using K constant product
    // tokensOut = currentTokens - (K / (currentBnb + bnbAmount))
    
    // Use mulDiv for precision if available (solidity 0.8.0+)
    uint256 newTokenReserve = K / (currentBnb + bnbAmount);
    
    // Handle edge case: if bnbAmount is tiny, newTokenReserve might equal currentTokens
    if (newTokenReserve >= currentTokens) {
        return 0; // Prevent underflow
    }
    
    return currentTokens - newTokenReserve;
}

        _burn(address(this), tokenDust);
    }
}

    require(amount > 0, "Invalid amount");

// Usage in buy():
function buy(uint256 minTokensOut) external payable nonReentrant {
    require(status == Status.Trading, "Trading is not active");
    require(msg.value > 0, "BNB amount must be positive");
    
    // Calculate fees
    uint256 platformFee = (msg.value * PLATFORM_FEE_BPS) / 10000;
    uint256 creatorFee = (msg.value * CREATOR_FEE_BPS) / 10000;
    uint256 bnbToReserve = msg.value - platformFee - creatorFee;
    
    // Calculate tokens
    uint256 tokensToMint = getAmountOfTokens(bnbToReserve);
    require(tokensToMint >= minTokensOut, "Slippage limit exceeded");
    
    // EFFECTS: Update state
    bnbReserves += bnbToReserve;
    _transfer(address(this), msg.sender, tokensToMint);
    
    if (bnbReserves >= GRADUATION_THRESHOLD_BNB) {
        status = Status.Graduated;
        emit GraduationTriggered(block.timestamp, bnbReserves, getCurrentPrice());
    }
    
    // INTERACTIONS: Send fees (after state changes)
    _sendBNB(platformFeeRecipient, platformFee);
    _sendBNB(creator, creatorFee);
    
    emit Buy(msg.sender, msg.value, tokensToMint, getCurrentPrice());
}

9. Known Limitations and Trade-offs
9.1 Bonding Curve Phase Limitations
Transaction Ordering Risk:

On busy networks, MEV bots could potentially front-run large purchases
Mitigation: Users should always use appropriate minTokensOut slippage protection
Note: This is inherent to all on-chain AMMs

Price Impact:

Large buys near graduation threshold will have significant price impact
Expected behavior: This is by design; late buyers pay premium prices

9.2 Gas Costs
Higher than PancakeSwap:

Bonding curve trades require complex math (constant product calculation)
Estimate: ~30-50% higher gas than standard AMM swap
Acceptable trade-off: This phase is temporary and provides fair launch mechanics

9.3 Centralization Vectors (Initial Version)
Platform admin can:
